local function sameTeam(plr)
    if not plr or plr == LocalPlayer then return true end
    local pT = plr:GetAttribute("TeamID")
    local lT = LocalPlayer:GetAttribute("TeamID")
    if pT ~= nil and lT ~= nil then return pT == lT end
    if plr.Team and LocalPlayer.Team then return plr.Team == LocalPlayer.Team end
    return false
end

local function getGunHL()
    if _gcHL and _gcHL.Parent then return _gcHL end
    local hl = Instance.new("Highlight")
    hl.Name = "FP_GunCham"
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Parent = CoreGui
    _gcHL = hl
    return hl
end

local function nameHit(list, n)
    n = n:lower()
    for k in pairs(list) do
        if n:find(k, 1, true) then return true end
    end
    return false
end

local function clearGunPaint()
    for p, data in pairs(_gcPainted) do
        pcall(function()
            if p and p.Parent and data then
                p.Color = data.Color
                p.Material = data.Material
                p.Transparency = data.Transparency
            end
        end)
        _gcPainted[p] = nil
    end
    if _gcHL then _gcHL.Enabled = false end
end

local function findGunModel()
    local cam = workspace.CurrentCamera
    if not cam then return nil end
    local lp = LocalPlayer
    local best, bestScore = nil, 0

    local function score(m)
        if not m or not m:IsA("Model") then return 0 end
        if lp and (m == lp.Character or m:IsDescendantOf(lp.Character)) then return 0 end
        local wpn, arms, parts = 0, 0, 0
        for _, d in ipairs(m:GetDescendants()) do
            if d:IsA("BasePart") and d.Transparency < 1 then
                parts += 1
                local n = d.Name
                if nameHit(ARM_NAME, n) then arms += 1
                elseif nameHit(WPN_NAME, n) then wpn += 1 end
            end
        end
        if parts == 0 then return 0 end
        -- pure-ish gun model scores high; mixed still ok if any weapon parts
        local s = wpn * 5 - arms * 2
        if wpn == 0 and arms == 0 and parts >= 2 then
            -- unnamed mesh gun — weak candidate
            s = parts * 0.5
        end
        local nm = m.Name:lower()
        if lp and nm:find(lp.Name:lower(), 1, true) then s += 2 end
        if nm:find("view", 1, true) or nm:find("gun", 1, true) or nm:find("weapon", 1, true) then s += 3 end
        return s
    end

    local function scan(root, depth)
        if not root or depth > 3 then return end
        for _, child in ipairs(root:GetChildren()) do
            if child:IsA("Model") then
                local sc = score(child)
                if sc > bestScore then bestScore, best = sc, child end
                scan(child, depth + 1)
            elseif child:IsA("Folder") then
                scan(child, depth + 1)
            end
        end
    end

    scan(cam, 0)
    scan(workspace:FindFirstChild("ViewModels"), 0)
    scan(workspace:FindFirstChild("ViewModel"), 0)
    scan(workspace:FindFirstChild("Camera"), 0)

    if bestScore >= 1 then return best end
    return nil
end

local function paintGun(model, fill, fillT)
    if not model then return end
    local seen = {}
    local painted = 0
    for _, d in ipairs(model:GetDescendants()) do
        if d:IsA("BasePart") and d.Transparency < 1 then
            local n = d.Name
            if nameHit(ARM_NAME, n) then continue end
            local ok = nameHit(WPN_NAME, n)
            -- if model is mostly weapon, also paint generic mesh parts
            if not ok then
                local ln = n:lower()
                if ln == "meshpart" or ln == "part" or ln == "handle" or ln:find("mesh") then
                    ok = true
                end
            end
            if not ok then continue end
            seen[d] = true
            painted += 1
            if not _gcPainted[d] then
                _gcPainted[d] = { Color = d.Color, Material = d.Material, Transparency = d.Transparency }
            end
            pcall(function()
                d.Color = fill
                if S.gc.Style == "Glass" then
                    d.Material = Enum.Material.Glass
                elseif S.gc.Style == "ForceField" then
                    d.Material = Enum.Material.ForceField
                else
                    d.Material = Enum.Material.Neon
                end
                d.Transparency = math.clamp(fillT, 0, 0.85)
            end)
        end
    end
    for p, data in pairs(_gcPainted) do
        if not seen[p] then
            pcall(function()
                if p.Parent and data then
                    p.Color = data.Color
                    p.Material = data.Material
                    p.Transparency = data.Transparency
                end
            end)
            _gcPainted[p] = nil
        end
    end
    -- highlight whole model as backup when few named parts
    local hl = getGunHL()
    if painted < 2 then
        hl.Adornee = model
        hl.FillColor = fill
        hl.OutlineColor = S.gc.OutlineColor or fill
        hl.FillTransparency = math.clamp(fillT, 0, 0.6)
        hl.OutlineTransparency = S.gc.OutT or 0
        hl.Enabled = true
    else
        hl.Enabled = false
    end
end

RunService.RenderStepped:Connect(function()
    if not S.gc or not S.gc.Enabled then
        clearGunPaint()
        return
    end
    local gun = findGunModel()
    if not gun then
        clearGunPaint()
        return
    end
    local sty = S.gc.Style
    local t = (math.sin(tick() * (S.gc.Speed or 4)) + 1) * 0.5
    local fill, fillT = S.gc.FillColor, S.gc.FillT
    if sty == "Hot" then
        fill = Color3.fromRGB(255, 90, 30):Lerp(Color3.fromRGB(255, 40, 200), t)
        fillT = 0.1
    elseif sty == "Rainbow" then
        fill = Color3.fromHSV((tick() * 0.25) % 1, 1, 1)
        fillT = 0.1
    elseif sty == "Glass" then
        fillT = 0.35
    elseif sty == "Blackout" then
        fill = Color3.new(0, 0, 0)
        fillT = 0
    elseif sty == "Pulse" then
        fillT = 0.05 + t * 0.35
    elseif sty == "ForceField" then
        fill = Color3.fromHSV((tick() * 0.15) % 1, 0.75, 1)
        fillT = 0.25
    end
    paintGun(gun, fill, fillT)
end)

local _vmOrig = {}
local function restoreVMPart(p)
    local o = _vmOrig[p]
    if not o then return end
    pcall(function()
        p.Color        = o.Color
        p.Material     = o.Material
        p.Transparency = o.Transparency
        p.Reflectance  = o.Reflectance
    end)
end
local function applyVM()
    local cam = workspace.CurrentCamera
    if not cam then return end
    local roots = { cam }
    local vmF = workspace:FindFirstChild("ViewModels")
    if vmF then table.insert(roots, vmF) end
    local seen = {}
    for _, root in ipairs(roots) do
        for _, d in ipairs(root:GetDescendants()) do
            if d:IsA("BasePart") then
                local n = d.Name:lower()
                local isVM = n:find("arm") or n:find("hand") or n:find("finger") or n:find("sleeve")
                    or n:find("glove") or n:find("weapon") or n:find("gun") or n:find("view")
                    or (d.Parent and d.Parent:IsA("Model") and d.Parent.Parent == cam)
                if not isVM and root == cam then isVM = true end
                if isVM then
                    seen[d] = true
                    if not _vmOrig[d] then
                        _vmOrig[d] = { Color = d.Color, Material = d.Material, Transparency = d.Transparency, Reflectance = d.Reflectance }
                    end
                    if S.vm.Enabled then
                        local t  = (math.sin(tick() * (S.vm.Speed or 3)) + 1) * 0.5
                        local sty = S.vm.Style
                        local col
                        if sty == "Rainbow" then
                            col = Color3.fromHSV((tick() * 0.2) % 1, 1, 1)
                        elseif sty == "Blackout" then
                            col = Color3.new(0, 0, 0)
                        else
                            col = S.vm.Color1:Lerp(S.vm.Color2, t)
                        end
                        if sty == "Hot" or sty == "Neon" or sty == "Rainbow" then
                            d.Material     = Enum.Material.Neon
                            d.Color        = col
                            d.Transparency = S.vm.Transparency
                            d.Reflectance  = 0.1
                        elseif sty == "ForceField" then
                            d.Material     = Enum.Material.ForceField
                            d.Color        = col
                            d.Transparency = S.vm.Transparency
                        elseif sty == "Glass" then
                            d.Material     = Enum.Material.Glass
                            d.Color        = col
                            d.Transparency = math.max(S.vm.Transparency, 0.25)
                            d.Reflectance  = 0.5
                        elseif sty == "Blackout" then
                            d.Material     = Enum.Material.SmoothPlastic
                            d.Color        = col
                            d.Transparency = S.vm.Transparency
                            d.Reflectance  = 0
                        else -- Smooth
                            d.Material     = Enum.Material.SmoothPlastic
                            d.Color        = col
                            d.Transparency = S.vm.Transparency
                        end
                    else
                        restoreVMPart(d)
                    end
                end
            end
        end
    end
    for p in pairs(_vmOrig) do
        if not seen[p] or not p.Parent then
            restoreVMPart(p)
            _vmOrig[p] = nil
        end
    end
end
RunService.RenderStepped:Connect(function() pcall(applyVM) end)

local function fx(class)
    local e = Lighting:FindFirstChildOfClass(class)
    if not e then
        e = Instance.new(class)
        e.Parent = Lighting
    end
    return e
end

local function applyWorld()
    local w = S.world
    if w.Fullbright then
        Lighting.Brightness = math.max(w.Brightness, 2)
        Lighting.Ambient = Color3.new(1, 1, 1)
        Lighting.GlobalShadows = false
    else
        Lighting.Brightness = w.Brightness
        if w.AmbOn then Lighting.Ambient = w.Ambient end
        Lighting.GlobalShadows = not w.NoShadows
    end
    pcall(function() Lighting.ExposureCompensation = w.Exposure end)
    if w.TimeOn then Lighting.ClockTime = w.Time end
    if w.NoFog then
        Lighting.FogStart = 0
        Lighting.FogEnd = 1000000
    elseif w.FogOn then
        Lighting.FogStart = w.FogStart
        Lighting.FogEnd = w.FogEnd
        Lighting.FogColor = w.FogColor
    end
    if w.AmbOn and not w.Fullbright then
        Lighting.Ambient = w.Ambient
        Lighting.OutdoorAmbient = w.OutdoorAmb
    end
    if w.ShiftOn then
        Lighting.ColorShift_Top = w.ShiftTop
        Lighting.ColorShift_Bottom = w.ShiftBot
    end
    local cam = workspace.CurrentCamera
    if cam then cam.FieldOfView = w.FOV end

    local bloom = fx("BloomEffect")
    bloom.Enabled = w.Bloom
    bloom.Intensity = w.BloomInt
    bloom.Size = w.BloomSize
    bloom.Threshold = w.BloomThresh

    local blur = fx("BlurEffect")
    blur.Enabled = w.Blur
    blur.Size = w.BlurSize

    local rays = fx("SunRaysEffect")
    rays.Enabled = w.Rays
    rays.Intensity = w.RaysInt
    rays.Spread = w.RaysSpread

    local cc = fx("ColorCorrectionEffect")
    cc.Enabled = w.CC
    cc.Brightness = 0
    cc.Contrast = w.Contrast
    cc.Saturation = w.Saturation
    cc.TintColor = w.TintColor

    local dof = fx("DepthOfFieldEffect")
    dof.Enabled = w.DOF
    dof.FarIntensity = w.DOFFar / 100
    dof.FocusDistance = w.DOFFocus * 200
    dof.InFocusRadius = w.DOFIn
    dof.NearIntensity = w.DOFOut / 100

    local at = Lighting:FindFirstChildOfClass("Atmosphere")
    if w.Atmos then
        if not at then at = Instance.new("Atmosphere"); at.Parent = Lighting end
        at.Density = w.AtmosDensity
        at.Offset = w.AtmosOffset
        at.Color = w.AtmosColor
        at.Decay = w.AtmosDecay
        at.Glare = w.AtmosGlare
        at.Haze = w.AtmosHaze
    elseif at then
        at:Destroy()
    end
end

local _lastShot = 0
local function markShot()
    _lastShot = tick()
    if S.sfx.Shoot then playSfx(S.sfx.ShootId) end
end
task.spawn(function()
    local rep = game:GetService("ReplicatedStorage")
    pcall(function()
        local UseItem = rep:WaitForChild("Remotes",10):WaitForChild("Replication",10):WaitForChild("Fighter",10):WaitForChild("UseItem",10)
        local EL = require(rep.Modules.EnumLibrary)
        local startEnum = EL:ToEnum("StartShooting")
        local old = UseItem.FireServer
        UseItem.FireServer = newcclosure(function(self, objID, enumVal, ...)
            local isShoot = (enumVal == startEnum)
            if not isShoot then local ss = tostring(enumVal or ""); isShoot = ss:find("StartShooting") or ss:find("Shooting") end
            if isShoot then markShot() end
            return old(self, objID, enumVal, ...)
        end)
    end)
    pcall(function()
        local oldNC; oldNC = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
            if getnamecallmethod() == "FireServer" then
                for _, a in ipairs({...}) do
                    local ss = tostring(a or "")
                    if ss:find("StartShooting") then markShot(); break end
                end
            end
            return oldNC(self, ...)
        end))
    end)
    UserInputService.InputBegan:Connect(function(inp, gpe)
        if gpe or inp.UserInputType ~= Enum.UserInputType.MouseButton1 then return end
        if not (S.sfx.Shoot or S.sfx.Hit or S.sfx.Kill) then return end
        local armed = false
        pcall(function()
            local fc = require(LocalPlayer.PlayerScripts.Controllers.FighterController)
            if fc and fc.LocalFighter and fc.LocalFighter.EquippedItem then armed = true end
        end)
        if armed or (LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")) then markShot() end
    end)
end)
local _hpCache = {}
RunService.Heartbeat:Connect(function()
    if not (S.sfx.Hit or S.sfx.Kill) then return end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer then continue end
        local hum = plr.Character and plr.Character:FindFirstChildOfClass("Humanoid")
        if not hum then continue end
        local id = plr.UserId
        local prev = _hpCache[id]; local hp = hum.Health
        if prev ~= nil and (tick()-_lastShot) < 0.8 then
            if S.sfx.Hit  and hp < prev-0.5 and hp > 0 then playSfx(S.sfx.HitId)  end
            if S.sfx.Kill and prev > 0        and hp <= 0  then playSfx(S.sfx.KillId) end
        end
        _hpCache[id] = hp
    end
end)

task.spawn(function()
    local ok, err = pcall(function()
        local rep = game:GetService("ReplicatedStorage")
        local plrs = game:GetService("Players")
        local runs = game:GetService("RunService")
        local cs = game:GetService("CollectionService")
        local lp = plrs.LocalPlayer
        local cam = workspace.CurrentCamera

        local U = require(rep.Modules.Utility)
        local EL = require(rep.Modules.EnumLibrary)
        local GU = require(rep.Modules.GameplayUtility)

        pcall(function()
            local ok2, gun = pcall(require, lp.PlayerScripts.Modules.ItemTypes.Gun)
            if ok2 and gun and gun.IsFullyAiming then
                local oldAim = gun.IsFullyAiming
                gun.IsFullyAiming = function(...)
                    if S.saim.Enabled then return true end
                    return oldAim(...)
                end
            end
        end)

        local FOVFill = Drawing.new("Circle")
        FOVFill.Filled = true
        FOVFill.NumSides = 64
        FOVFill.ZIndex = 1
        FOVFill.Visible = false

        local FOV = Drawing.new("Circle")
        FOV.Filled = false
        FOV.NumSides = 64
        FOV.Thickness = 1
        FOV.ZIndex = 3
        FOV.Visible = false

        local Snap = Drawing.new("Line")
        Snap.Thickness = 1
        Snap.ZIndex = 3
        Snap.Visible = false

        local function hideAllFov()
            FOVFill.Visible = false
            FOV.Visible = false
            Snap.Visible = false
        end

        local function target()
            if not S.saim.Enabled then return nil end
            cam = workspace.CurrentCamera
            if not cam then return nil end

            local center = cam.ViewportSize / 2
            local fov = (getgenv().Config and getgenv().Config.FOVRadius) or S.saim.FOVRadius or 300
            local hitName = (getgenv().Config and getgenv().Config.HitPart) or S.saim.HitPart or "Head"
            local myRoot = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
            local origin = myRoot and myRoot.Position or cam.CFrame.Position
            local bestPart, bestDist = nil, fov

            local function teamOk(plr)
                if not S.saim.TeamCheck then return true end
                local pT = plr:GetAttribute("TeamID")
                local lT = lp:GetAttribute("TeamID")
                if pT ~= nil and lT ~= nil then return pT ~= lT end
                if plr.Team and lp.Team then return plr.Team ~= lp.Team end
                return true
            end

            local function resolveModel(plr)
                -- rivals body is often an Entity model, not Character
                for _, e in ipairs(cs:GetTagged("Entity")) do
                    if typeof(e) == "Instance" and e:IsA("Model") then
                        local n = e.Name
                        if n == plr.Name or n:find(plr.Name, 1, true) then
                            return e
                        end
                        local attr = e:GetAttribute("UserId") or e:GetAttribute("PlayerUserId")
                        if attr and tonumber(attr) == plr.UserId then
                            return e
                        end
                    end
                end
                if plr.Character and plr.Character.Parent then
                    return plr.Character
                end
                return nil
            end

            local function alive(model, plr)
                local hum = model:FindFirstChildOfClass("Humanoid")
                if hum then return hum.Health > 0 end
                if plr.Character then
                    local h2 = plr.Character:FindFirstChildOfClass("Humanoid")
                    if h2 then return h2.Health > 0 end
                end
                return model:FindFirstChild("Head") ~= nil
            end

            local function losOk(part)
                if not S.saim.WallCheck then return true end
                local params = RaycastParams.new()
                params.FilterType = Enum.RaycastFilterType.Exclude
                params.FilterDescendantsInstances = { lp.Character, cam }
                params.IgnoreWater = true
                local dir = part.Position - origin
                if dir.Magnitude < 1 then return true end
                local hit = workspace:Raycast(origin, dir, params)
                if not hit then return true end
                return hit.Instance and hit.Instance:IsDescendantOf(part.Parent)
            end

            local function getHitPart(model)
                local part = model:FindFirstChild(hitName, true)
                if part and part:IsA("BasePart") then return part end
                for _, n in ipairs({ "Head", "UpperTorso", "HumanoidRootPart", "Torso" }) do
                    local p = model:FindFirstChild(n, true)
                    if p and p:IsA("BasePart") then return p end
                end
                return nil
            end

            local fov = (getgenv().Config and getgenv().Config.FOVRadius) or S.saim.FOVRadius or 300
            bestPart, bestDist = nil, math.huge

            for _, plr in ipairs(plrs:GetPlayers()) do
                if plr == lp then continue end
                if not teamOk(plr) then continue end
                local model = resolveModel(plr)
                if not model then continue end
                if not alive(model, plr) then continue end
                local part = getHitPart(model)
                if not part then continue end
                if part.Transparency >= 0.99 then continue end
                local worldDist = (part.Position - origin).Magnitude
                if worldDist > (S.saim.MaxDist or 1500) then continue end
                if not losOk(part) then continue end

                local sp = cam:WorldToViewportPoint(part.Position)
                if sp.Z <= 0 then continue end
                local screenDist = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                if screenDist > fov then continue end

                if worldDist < bestDist then
                    bestDist = worldDist
                    bestPart = part
                end
            end

            return bestPart
        end

        local function CFD(origin, part)
            local aimPos = part.Position
            local look = CFrame.lookAt(origin, aimPos)
            local cf = part.CFrame
            local d = {}
            d[utf8.char(1)] = {
                [utf8.char(0)] = U:EncodeCFrame(look),
                [utf8.char(1)] = U:EncodeCFrame(S.saim.Manipulate and look or cf),
                [utf8.char(2)] = part,
                [utf8.char(3)] = U:EncodeCFrame(cf:ToObjectSpace(CFrame.new(aimPos))),
            }
            return d
        end

        local lastTP = nil

        runs.RenderStepped:Connect(function()
            cam = workspace.CurrentCamera
            if not cam then hideAllFov() lastTP = nil return end
            if not (S.saim.Enabled and S.saim.ShowFOV) then
                hideAllFov()
                lastTP = nil
                return
            end

            local t = target()
            lastTP = t and t.Position or nil

            local radius = (getgenv().Config and getgenv().Config.FOVRadius) or S.saim.FOVRadius or 300
            if S.saim.Anim then
                radius = radius + math.sin(tick() * (S.saim.PulseSpeed or 4)) * 6
            end

            local center = cam.ViewportSize / 2
            local pos = center
            if S.saim.FollowTarget and lastTP then
                local sp, on = cam:WorldToViewportPoint(lastTP)
                if on and sp.Z > 0 then pos = Vector2.new(sp.X, sp.Y) end
            end

            local hue = (tick() * 0.35) % 1
            local col = S.saim.FOVColor or Color3.new(1,1,1)
            if S.saim.Rainbow then col = Color3.fromHSV(hue, 1, 1) end
            local fillCol = S.saim.FOVFillColor or Color3.fromRGB(255, 80, 120)
            if S.saim.Rainbow then fillCol = Color3.fromHSV((hue + 0.15) % 1, 0.9, 1) end
            local fillA = math.clamp(S.saim.FOVFillAlpha or 0.65, 0, 1)
            local sides = S.saim.Sides or 64
            local thick = S.saim.Thickness or 1

            FOVFill.Position = pos
            FOVFill.Radius = radius
            FOVFill.Color = fillCol
            FOVFill.Transparency = fillA
            FOVFill.NumSides = sides
            FOVFill.Visible = S.saim.FOVFill == true

            FOV.Position = pos
            FOV.Radius = radius
            FOV.Color = col
            FOV.Thickness = thick
            FOV.NumSides = sides
            FOV.Visible = true

            if S.saim.Snapline and lastTP then
                local sp, on = cam:WorldToViewportPoint(lastTP)
                if on and sp.Z > 0 then
                    Snap.From = center
                    Snap.To = Vector2.new(sp.X, sp.Y)
                    Snap.Color = col
                    Snap.Thickness = thick
                    Snap.Visible = true
                else
                    Snap.Visible = false
                end
            else
                Snap.Visible = false
            end
        end)

        local visual = GU.GetEntitiesFromRaycast
        GU.GetEntitiesFromRaycast = function(self, envID, params, origin, dir, maxDist, ...)
            if S.saim.Enabled then
                local t = target()
                if t then
                    local dist = (t.Position - origin).Magnitude
                    dir = (t.Position - origin).Unit
                    if dist > maxDist then maxDist = dist + 5 end
                end
            end
            return visual(self, envID, params, origin, dir, maxDist, ...)
        end

        local UseItem = rep.Remotes.Replication.Fighter.UseItem
        local startEnum = EL:ToEnum("StartShooting")
        local oldFS
        oldFS = hookfunction(UseItem.FireServer, newcclosure(function(self, objID, enumVal, camdata, extra)
            if S.saim.Enabled and enumVal == startEnum then
                local chance = S.saim.HitChance or 100
                if chance >= 100 or math.random(1, 100) <= chance then
                    local root = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
                    local t = target()
                    if root and t then
                        camdata = CFD(root.Position, t)
                    end
                end
            end
            if S.tracer2 and S.tracer2.Enabled and enumVal == startEnum then
                pcall(function()
                    local c = workspace.CurrentCamera
                    if not c then return end
                    local char = lp.Character
                    local from = c.CFrame.Position + c.CFrame.LookVector * 1.5
                    local to = nil
                    if type(camdata) == "table" then
                        local pack = camdata[utf8.char(1)]
                        if type(pack) == "table" then
                            local part = pack[utf8.char(2)]
                            if typeof(part) == "Instance" and part:IsA("BasePart") then
                                to = part.Position
                            end
                        end
                    end
                    if not to then
                        local params = RaycastParams.new()
                        params.FilterType = Enum.RaycastFilterType.Exclude
                        params.FilterDescendantsInstances = { char, c }
                        local hit = workspace:Raycast(c.CFrame.Position, c.CFrame.LookVector * 900, params)
                        to = hit and hit.Position or (c.CFrame.Position + c.CFrame.LookVector * 400)
                    end
                    if getgenv().FP_spawnTracer then getgenv().FP_spawnTracer(from, to) end
                end)
            end
            return oldFS(self, objID, enumVal, camdata, extra)
        end))
    end)
end)

            end
        end

    local function patchItem(item)
        if not item then return end
        pcall(function()
            local info = rawget(item,"Info") or rawget(item,"_data")
            if type(info) == "table" then
                if S.gun.NoRecoil then
                    for _, k in ipairs({"ShootRecoil","Recoil","CameraKick","RecoilX","RecoilY"}) do
                        if info[k] ~= nil then info[k] = 0 end
                    end
                end
                if S.gun.NoSpread then
                    for _, k in ipairs({"Spread","BaseSpread","SpreadMagnitude","BulletSpread"}) do
                        if info[k] ~= nil then info[k] = 0 end
                    end
                end
                if S.gun.FastReload then
                    for _, k in ipairs({"ReloadTime","EmptyReload","ReloadSpeed"}) do
                        if info[k] ~= nil then info[k] = 0.05 end
                    end
                end
                if S.gun.FireRate then
                    local m = S.gun.FireRateMul or 0.15
                    for _, k in ipairs({"ShootCooldown","FireCooldown","AttackCooldown"}) do
                        if type(info[k]) == "number" then
                            info["_base_"..k] = info["_base_"..k] or info[k]
                            info[k] = info["_base_"..k] * m
                        end
                    end
                end
            end
        end)
    end

    while true do
        if S.gun.NoRecoil or S.gun.NoSpread or S.gun.FastReload or S.gun.FireRate then
            pcall(function()
                local fc = require(LocalPlayer.PlayerScripts.Controllers.FighterController)
                local lf = fc and fc.LocalFighter
                if lf and lf.EquippedItem then patchItem(lf.EquippedItem) end
            end)
            if getgc then
                for _, obj in ipairs(getgc(true)) do
                    if type(obj) == "table" then
                        pcall(function()
                            if S.gun.NoSpread then
                                if rawget(obj,"SpreadMagnitude") then rawset(obj,"SpreadMagnitude",0) end
                                if rawget(obj,"BaseSpread")      then rawset(obj,"BaseSpread",0)      end
                                if rawget(obj,"BulletSpread")    then rawset(obj,"BulletSpread",0)    end
                            end
                            if S.gun.NoRecoil then
                                if rawget(obj,"ShootRecoil") then rawset(obj,"ShootRecoil",0) end
                                if rawget(obj,"CameraKick")  then rawset(obj,"CameraKick",0)  end
                            end
                        end)
                    end
                end
            end
        end
        task.wait(0.2)
    end
end)

local function applyMove()
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    if S.move.Noclip then
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") then p.CanCollide = false end
        end
    end

    if S.move.Jump then
        hum.JumpPower = S.move.JumpVal
        pcall(function() hum.JumpHeight = S.move.JumpVal / 5 end)
    end

    local want = nil
    if S.move.Sprint and _sprint then
        want = S.move.SprintVal
    elseif S.move.Speed then
        want = S.move.SpeedVal
    end
    if want and not S.move.Fly then
        if math.abs(hum.WalkSpeed - want) > 0.1 then
            hum.WalkSpeed = want
        end
    end

    if S.move.Fly then
        local cam = workspace.CurrentCamera
        local dir = Vector3.zero
        if cam then
            local look = cam.CFrame.LookVector
            local right = cam.CFrame.RightVector
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir += look end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir -= look end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir -= right end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir += right end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.yAxis end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then dir -= Vector3.yAxis end
        end
        if dir.Magnitude > 0 then
            hrp.AssemblyLinearVelocity = dir.Unit * S.move.FlySpeed
        else
            hrp.AssemblyLinearVelocity = Vector3.zero
        end
        hum.PlatformStand = false
    end
end

local function refreshEntityMap()
    table.clear(_entityByPlayer)
    table.clear(_playerByModel)
    local FC = getFighterController()
    if FC and FC.Objects then
        for _, fo in pairs(FC.Objects) do
            pcall(function()
                local plr = fo.Player
                local ent = fo.Entity
                local model = ent and (ent.Model or (ent.GetModel and ent:GetModel()))
                if not model and fo.Get and fo:Get("Model") then model = fo:Get("Model") end
                if not model and typeof(fo.Model) == "Instance" then model = fo.Model end
                if plr and model then
                    _entityByPlayer[plr] = model
                    _playerByModel[model] = plr
                end
            end)
        end
    end
    pcall(function()
        for _, model in ipairs(CollectionService:GetTagged("Entity")) do
            if model:IsA("Model") then
                local plr = _playerByModel[model]
                if not plr then
                    for _, p in ipairs(Players:GetPlayers()) do
                        if p ~= LocalPlayer and p.Character == model then
                            plr = p
                            break
                        end
                        if p ~= LocalPlayer and (model.Name == p.Name or model:GetAttribute("PlayerName") == p.Name) then
                            plr = p
                            break
                        end
                    end
                end
                if plr then
                    _entityByPlayer[plr] = model
                    _playerByModel[model] = plr
                end
            end
        end
    end)
end

local function isVisible(origin, char)
    local tgt = char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")

end

local function getState(hum)
    if not hum then return "Standing" end
    local st=hum:GetState()
    if st==Enum.HumanoidStateType.Running then return "Walking"  end
    if st==Enum.HumanoidStateType.Jumping  then return "Jumping"  end
    if st==Enum.HumanoidStateType.Freefall then return "Falling"  end
    if st==Enum.HumanoidStateType.Climbing then return "Climbing" end
    if st==Enum.HumanoidStateType.Swimming then return "Swimming" end
    return "Standing"
end

RunService.RenderStepped:Connect(function()
    _espFrame += 1

        if _espFrame % 10 == 0 then
            for _, sv in pairs(sets) do hideAll(sv) end
        end
        return
    end

        if not char then continue end

                local corners = {
                    pos + Vector3.new( half.X,  half.Y,  half.Z),
                    pos + Vector3.new(-half.X,  half.Y,  half.Z),
                    pos + Vector3.new( half.X, -half.Y,  half.Z),
                    pos + Vector3.new(-half.X, -half.Y,  half.Z),
                    pos + Vector3.new( half.X,  half.Y, -half.Z),
                    pos + Vector3.new(-half.X,  half.Y, -half.Z),
                    pos + Vector3.new( half.X, -half.Y, -half.Z),
                    pos + Vector3.new(-half.X, -half.Y, -half.Z),
                }
                for _, c in ipairs(corners) do
                    local sp = cam:WorldToViewportPoint(c)
                    if sp.Z > 0 then
                        anyOn = true
                        if sp.X < minX then minX = sp.X end
                        if sp.Y < minY then minY = sp.Y end
                        if sp.X > maxX then maxX = sp.X end
                        if sp.Y > maxY then maxY = sp.Y end
                    end
                end
            end
        end
        if not anyOn then continue end

        local y = minY - pad

            if p:sub(1,5)=="Right"  then return "Right"  end

        end
        local function snap(pos, label, lw, lh)
            local side=normP(pos); local cur=cursor[side] or (y-2); local px,py
            if side=="Top"    then py=cur-lh; px=math.floor(x+w*0.5-lw*0.5); cursor[side]=py-1
            elseif side=="Bottom" then py=cur; px=math.floor(x+w*0.5-lw*0.5); cursor[side]=cur+lh+1
            elseif side=="Left"   then py=math.floor(cur); px=math.floor(x-lw-4); cursor[side]=cur+lh+1
            elseif side=="Right"  then py=math.floor(cur); px=math.floor(x+w+4); cursor[side]=cur+lh+1
            else py=math.floor(cur); px=math.floor(x+w*0.5-lw*0.5); cursor[side]=cur+lh+1 end
            label.Position=UDim2.fromOffset(px,math.floor(py)); label.Size=UDim2.fromOffset(lw,lh)
            label.TextSize=S.TextSize or 10; label.TextTransparency=1-fade
        end

        if S.Box then
            s.box.Position=UDim2.fromOffset(math.floor(x),math.floor(y))
            s.box.Size=UDim2.fromOffset(math.floor(w),math.floor(h))
            if S.BoxStyle=="Corner" then

                s.stroke.Thickness=math.max(S.BoxThickness, 1); s.stroke.Transparency=1-fade; s.box.Visible=true
            elseif S.BoxStyle=="Full" then
                s.stroke.Color=S.TeamColor and plr.TeamColor and plr.TeamColor.Color or S.BoxColor
                s.stroke.Thickness=math.max(S.BoxThickness, 1); s.stroke.Transparency=1-fade; s.box.Visible=true
            else -- 3D handled by stroke as well
                s.stroke.Color=S.BoxColor; s.stroke.Thickness=S.BoxThickness; s.stroke.Transparency=1-fade; s.box.Visible=true
            end
        else s.box.Visible=false end

        if S.BoxFill and S.Box then
            s.fill.Size=UDim2.fromScale(1,1); s.fill.BackgroundColor3=S.BoxFillColor; s.fill.BackgroundTransparency=S.BoxFillAlpha; s.fill.Visible=true

            s.hpFill.Size=UDim2.fromOffset(3,math.floor(h*frac)); s.hpFill.Position=UDim2.fromOffset(0,math.floor(h*(1-frac)))

        if S.Username then s.user.Text="@"..plr.Name; s.user.Visible=true; snap(S.NamePos,s.user,70,8) else s.user.Visible=false end

        if S.Distance then s.dist.Text=string.format("%.0fm",dist); s.dist.Visible=true; snap(S.DistPos,s.dist,30,10) else s.dist.Visible=false end

        if S.Chams then

            pcall(function()

        end

        if S.Aura then
            if not s.aura then
                s.aura=Instance.new("ParticleEmitter"); s.aura.Texture="rbxassetid://243098098"
                s.aura.Rate=20; s.aura.Lifetime=NumberRange.new(0.5,1); s.aura.Speed=NumberRange.new(1,3)

            s.aura.Color=ColorSequence.new(S.ChamFill); s.aura.Enabled=true
        elseif s.aura then s.aura.Enabled=false end
    end
    for char, sv in pairs(sets) do if not seen[char] then hideAll(sv) end end
end)

local _beams = {}
local _lastShot = 0

getgenv().FP_spawnTracer = function(from, to)
    if not S.tracer2.Enabled then return end
    if typeof(from) ~= "Vector3" or typeof(to) ~= "Vector3" then return end
    local now = tick()
    if now - _lastShot < 0.05 then return end
    _lastShot = now

    local a0 = Instance.new("Attachment")
    a0.WorldPosition = from
    a0.Parent = workspace.Terrain

    local a1 = Instance.new("Attachment")
    a1.WorldPosition = to
    a1.Parent = workspace.Terrain

    local beam = Instance.new("Beam")
    beam.Attachment0 = a0
    beam.Attachment1 = a1
    beam.FaceCamera = true
    beam.LightEmission = S.tracer2.Emission or 1
    beam.Brightness = S.tracer2.Brightness or 1
    beam.Width0 = S.tracer2.Width
    beam.Width1 = S.tracer2.Width
    beam.Color = ColorSequence.new(S.tracer2.Color)
    beam.Transparency = NumberSequence.new(0)

    local tex = TRACER_TEX[S.tracer2.Style]
    if tex and tex ~= "" then
        beam.Texture = tex
        beam.TextureLength = 4
        beam.TextureSpeed = 1
        beam.TextureMode = Enum.TextureMode.Wrap
    end

    beam.Parent = workspace.Terrain
    table.insert(_beams, {
        beam = beam, a0 = a0, a1 = a1,
        t = now,
        life = S.tracer2.Lifetime or 0.55,
        fade = S.tracer2.FadeTime or 0.3,
        w = S.tracer2.Width,
        expand = S.tracer2.Expand,
    })
end

RunService.RenderStepped:Connect(function()
    local now = tick()
    for i = #_beams, 1, -1 do
        local b = _beams[i]
        local age = now - b.t
        if age >= b.life + b.fade then
            pcall(function() b.beam:Destroy() end)
            pcall(function() b.a0:Destroy() end)
            pcall(function() b.a1:Destroy() end)
            table.remove(_beams, i)
        elseif age >= b.life then
            local f = (age - b.life) / math.max(b.fade, 0.001)
            pcall(function()
                b.beam.Transparency = NumberSequence.new(math.clamp(f, 0, 1))
                if b.expand then
                    b.beam.Width0 = b.w * (1 + f)
                    b.beam.Width1 = b.w * (1 + f) * 0.5
                end
            end)
        end
    end
end)
